on_game_start = {
    on_actions = { crypt_balance_byzantium_feudal_start }
}
on_game_start_after_lobby = {
    on_actions = { crypt_balance_byzantium_feudal_start }
}

crypt_balance_byzantium_feudal_start = {
    trigger = {
        game_start_date = 867.1.1
        has_game_rule = crypt_balance_byzantium_867_feudal
    }

    effect = {
        title:e_byzantium.holder ?= {
            if = {
                limit = { has_government = administrative_government }

                # Snapshot vassals and family titles before changing the liege's government.
                every_vassal_or_below = {
                    limit = { has_government = administrative_government is_landed = yes }
                    add_to_temporary_list = crypt_balance_byzantine_governors
                    empty_treasury_when_abandoning_landed_life_effect = yes
                    if = {
                        limit = { is_governor = yes has_trait = governor }
                        remove_trait = governor
                    }
                }
                every_noble_family = {
                    add_to_temporary_list = crypt_balance_byzantine_estates
                }
                every_held_title = {
                    limit = {
                        is_noble_family_title = yes
                        NOT = { is_in_list = crypt_balance_byzantine_estates }
                    }
                    add_to_temporary_list = crypt_balance_byzantine_estates
                }

                empty_treasury_when_abandoning_landed_life_effect = yes
                change_government = feudal_government
                every_in_list = {
                    list = crypt_balance_byzantine_governors
                    limit = { has_government = administrative_government }
                    change_government = feudal_government
                }
                every_in_list = {
                    list = crypt_balance_byzantine_estates
                    holder ?= { destroy_title = prev }
                }
            }

            # The post-lobby pass also covers a replacement ruler from Ruler Designer.
            if = {
                limit = { has_government = feudal_government }
                if = {
                    limit = { NOT = { has_realm_law = single_heir_succession_law } }
                    add_realm_law_skip_effects = single_heir_succession_law
                }
                crypt_balance_enforce_succession_effect = yes
            }
        }
    }
}

crypt_balance_apply_adventurer_opinion_effect = {
	save_temporary_scope_as = crypt_balance_adventurer
	set_variable = crypt_balance_adventurer_opinion_active
	every_ruler = {
		limit = { is_landed = yes }
		crypt_balance_sync_opinion_of_adventurer_effect = { ADVENTURER = scope:crypt_balance_adventurer }
	}
}

crypt_balance_sync_opinion_of_adventurer_effect = {
	if = {
		limit = {
			is_landed = yes
			NOT = { has_opinion_modifier = { target = $ADVENTURER$ modifier = crypt_balance_landed_opinion_of_adventurer } }
		}
		add_opinion = { target = $ADVENTURER$ modifier = crypt_balance_landed_opinion_of_adventurer }
	}
	else_if = {
		limit = {
			is_landed = no
			has_opinion_modifier = { target = $ADVENTURER$ modifier = crypt_balance_landed_opinion_of_adventurer }
		}
		remove_opinion = { target = $ADVENTURER$ modifier = crypt_balance_landed_opinion_of_adventurer }
	}
}

# Province scope. Replace the previous population bonus when county prosperity changes.
crypt_balance_update_prosperity_population_effect = {
	save_temporary_scope_value_as = {
		name = crypt_balance_settlement_prosperity_level
		value = {
			value = county.prosperity_level
			floor = yes
			min = 0
			max = 100
		}
	}
	if = {
		limit = {
			NOT = { has_variable = crypt_balance_applied_settlement_prosperity }
		}
		crypt_balance_remove_legacy_settlement_prosperity_effect = yes
		set_variable = { name = crypt_balance_applied_settlement_prosperity value = 0 }
	}
	if = {
		limit = {
			NOT = { var:crypt_balance_applied_settlement_prosperity = scope:crypt_balance_settlement_prosperity_level }
		}
		crypt_balance_change_settlement_prosperity_modifier_effect = {
			LEVEL = var:crypt_balance_applied_settlement_prosperity
			OPERATION = remove_province_modifier
		}
		crypt_balance_change_settlement_prosperity_modifier_effect = {
			LEVEL = scope:crypt_balance_settlement_prosperity_level
			OPERATION = add_province_modifier
		}
		set_variable = {
			name = crypt_balance_applied_settlement_prosperity
			value = scope:crypt_balance_settlement_prosperity_level
		}
	}
}

crypt_balance_remove_legacy_settlement_prosperity_effect = {
